# half lives is a choice rather than a convention to find. Here it is one file
# per project under ~/.age/, which is the shape the rest of this machine's
# secret tooling already uses, and which nothing publishes:
#
#   mkdir -p ~/.age && chmod 700 ~/.age
#   age-keygen -o ~/.age/isometric-prototype.txt   # prints the age1… line to add below
#
#   AGE_IDENTITY_FILE=~/.age/isometric-prototype.txt scripts/art-secrets.sh decrypt
#
# CI gets the same key as the AGE_KEY secret, set once:
#
#   gh secret set AGE_KEY < ~/.age/isometric-prototype.txt
